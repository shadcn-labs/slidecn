## Usage

```tsx
import Callout from "@/components/slides/callout";

function MyPresentation() {
  return (
    <Deck>
      <Callout variant="warning" title="Caution">
        This operation cannot be undone.
      </Callout>
    </Deck>
  );
}
```

## API Reference

### Callout

| Prop       | Type                                       | Default  | Description     |
| ---------- | ------------------------------------------ | -------- | --------------- |
| `variant`  | `"info" \| "warning" \| "tip" \| "danger"` | `"info"` | Callout style   |
| `title`    | `string`                                   | -        | Callout heading |
| `children` | `ReactNode`                                | required | Callout content |
| `icon`     | `boolean`                                  | `true`   | Show icon       |
